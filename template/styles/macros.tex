% CABEÇALHO DE ARTIGO

\newcommand{\articleheader}[2]{

\begin{tcolorbox}[
colback=HypeOrange,
colframe=HypeOrange,
arc=0mm,
boxrule=0pt,
left=3mm,
right=3mm,
top=2.5mm,
bottom=2.5mm,
halign=left,
]

{\color{white}
\opensansfont
\raggedright
\fontsize{18}{20}\selectfont
\bfseries
#1\par
}

\end{tcolorbox}
{\small\color{HypePurple} Texto por #2}

\vspace{0.2cm}

}

% MACRO: ARTIGO -----------------------------------------

\newcommand{\hypearticle}[3]{

\vspace{0.1cm}

\articleheader{#1}{#2}

#3

\vspace{0.5cm}

}

% MACRO: SUBTÍTULO -------------------------------------

\newcommand{\subtitle}[1]{

\vspace{0.4cm}

{\large\bfseries\color{HypeBlue} #1}

}

% MACRO: BLOCO DE DESTAQUE -----------------------------

\newcommand{\highlight}[1]{

\vspace{0.3cm}

\noindent
\colorbox{HypePurple!20}{
\parbox{0.95\columnwidth}{
\vspace{0.15cm}
#1
\vspace{0.15cm}
}
}

}

% MACRO: FONTE ORIGINAL --------------------------------

\newcommand{\originalsource}[1]{

\vspace{0.15cm}


{\color{HypePurple2}Gostou do texto?}\\
Leia também a fonte original
\href{#1}{aqui}.

}

% IMAGEM ------------------------------------------------

\newcommand{\articleimage}[3]{

\begin{center}

\includegraphics[width=#1\columnwidth]{#2}

\end{center}
{\small\color{gray}Legenda: #3}

}

\newcommand{\summaryitem}[4]{
\begin{tcolorbox}[
    colback=white,
    colframe=white,
    boxrule=0pt,
    arc=4mm,
    left=4mm,
    right=4mm,
    top=3mm,
    bottom=3mm
]

\begin{minipage}[c]{0.12\linewidth}
    {\fontsize{28}{28}\selectfont
    \color{HypeOrange}\textbf{#1}}
\end{minipage}
\hfill
\begin{minipage}[c]{0.68\linewidth}
    {\Large\color{HypePurple}\textbf{#2}}

    \vspace{0.15cm}

    #3
\end{minipage}
\hfill
\begin{minipage}[c]{0.12\linewidth}
    \raggedleft
    {\fontsize{28}{28}\selectfont
    \color{HypePurple}\textbf{#4}}

    {\small\color{HypeOrange}\textbf{pág.}}
\end{minipage}

\end{tcolorbox}

\vspace{0.3cm}
}
